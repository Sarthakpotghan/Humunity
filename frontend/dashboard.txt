        modal.classList.remove('opacity-0', 'pointer-events-none');
      } else {
        modal.classList.add('opacity-0', 'pointer-events-none');
      }
    }

    if (openCreateBtn) openCreateBtn.addEventListener('click', () => toggleModal(createModal, true));
    if (closeCreateBtn) closeCreateBtn.addEventListener('click', () => toggleModal(createModal, false));
    if (cancelCreateBtn) cancelCreateBtn.addEventListener('click', () => toggleModal(createModal, false));

    if (triggerAcceptModal) triggerAcceptModal.addEventListener('click', () => toggleModal(acceptModal, true));
    if (closeAcceptBtn) closeAcceptBtn.addEventListener('click', () => toggleModal(acceptModal, false));
    if (cancelAcceptBtn) cancelAcceptBtn.addEventListener('click', () => toggleModal(acceptModal, false));

    if (confirmAcceptBtn) {
      confirmAcceptBtn.addEventListener('click', () => {
        toggleModal(acceptModal, false);
        showToast('PATCH /matches/409/accept processed. Courier dispatched!');
        if (triggerAcceptModal) {
          triggerAcceptModal.innerHTML = '<span class="material-symbols-outlined text-[18px]">verified</span> Matched &amp; Accepted';
          triggerAcceptModal.classList.remove('bg-tertiary', 'text-on-tertiary');
          triggerAcceptModal.classList.add('bg-surface-container-high', 'text-on-surface-variant', 'pointer-events-none');
        }
      });
    }

    window.submitNeedForm = function() {
      toggleModal(createModal, false);
      showToast('Need registered (POST /requests). Community network notified.');
      document.getElementById('resourceNeedForm').reset();
    };

    const urgencyInput = document.getElementById('urgencyRangeInput');
    const urgencyVal = document.getElementById('urgencyValDisplay');
    const urgencyBadge = document.getElementById('urgencyBadge');

    if (urgencyInput && urgencyVal && urgencyBadge) {
      urgencyInput.addEventListener('input', (e) => {
        const val = e.target.value;
        urgencyVal.textContent = val;
        if (val == 5) {
          urgencyBadge.className = 'font-label-sm text-label-sm px-space-sm py-0.5 rounded-full bg-error-container text-on-error-container';
          urgencyBadge.textContent = 'Level 5 - Critical Immediate';
        } else if (val == 4) {
          urgencyBadge.className = 'font-label-sm text-label-sm px-space-sm py-0.5 rounded-full bg-secondary-fixed text-on-secondary-fixed-variant';
          urgencyBadge.textContent = 'Level 4 - High Urgency';
        } else if (val == 3) {
          urgencyBadge.className = 'font-label-sm text-label-sm px-space-sm py-0.5 rounded-full bg-secondary-container text-on-secondary-container';
          urgencyBadge.textContent = 'Level 3 - Pressing Need';
        } else {
          urgencyBadge.className = 'font-label-sm text-label-sm px-space-sm py-0.5 rounded-full bg-surface-container-high text-on-surface-variant';
          urgencyBadge.textContent = 'Level ' + val + ' - Standard Intake';
        }
      });
    }

    const filterBtns = document.querySelectorAll('.filter-tab-btn');
    const rows = document.querySelectorAll('.table-row-item');

    filterBtns.forEach(btn => {
      btn.addEventListener('click', () => {
        filterBtns.forEach(b => {
          b.classList.remove('bg-on-surface', 'text-surface-container-lowest');
          b.classList.add('text-on-surface-variant');
        });
        btn.classList.add('bg-on-surface', 'text-surface-container-lowest');
        btn.classList.remove('text-on-surface-variant');

        const filter = btn.getAttribute('data-filter');
        rows.forEach(row => {
          const status = row.getAttribute('data-status');
          if (filter === 'all' || status === filter) {
            row.style.display = '';
          } else {
            row.style.display = 'none';
          }
        });
      });
    });
  })();
